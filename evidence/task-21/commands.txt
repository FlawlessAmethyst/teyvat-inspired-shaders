import csv
from statistics import mean
import sys

with open(sys.argv[1], newline='') as file:
    rows = list(csv.DictReader(file))
for scene in ('daylight', 'rain'):
    values = {}
    for mode in ('sodium', 'control', 'weather'):
        runs = [row for row in rows if row['scene'] == scene and row['batch'].startswith(mode)]
        assert len(runs) == 2
        values[mode] = [float(row['mean_client_loop_ms']) for row in runs]
        p95 = [float(row['p95_client_loop_ms']) for row in runs]
        print(f'{scene} {mode}: mean {mean(values[mode]):.6f} ms, runs {values[mode]}, p95 {p95}')
    differences = [a - b for a, b in zip(values['weather'], values['control'], strict=True)]
    print(f'{scene} TASK-21 minus TASK-20: {mean(differences):+.6f} ms '
          f'({mean(differences) / mean(values["control"]):+.2%}), runs {differences}')
PY

# Link all pairs and expand the sky sources for the existing isolated GPU timer.
python3 - "$root" <<'PY' > "$root/static-link.txt"
from pathlib import Path
import subprocess
import sys
from tempfile import TemporaryDirectory

sys.path.insert(0, 'evidence/task-12')
from check_glsl import expand

stages = {'SKY': 1, 'SUNSET': 2, 'STARS': 6, 'HAND_SOLID': 16, 'CLOUDS': 20,
          'RAIN_SNOW': 21, 'HAND_TRANSLUCENT': 23}
defines = [f'-DMC_RENDER_STAGE_{name}={value}' for name, value in stages.items()]
root = Path(sys.argv[1])
shaders = root / 'shaders'
with TemporaryDirectory(prefix='task-21-glsl-') as temp:
    for fragment in sorted(shaders.glob('*.fsh')):
        vertex = fragment.with_suffix('.vsh')
        vert = Path(temp) / (vertex.stem + '.vert')
        frag = Path(temp) / (fragment.stem + '.frag')
        vert.write_text(expand(vertex, shaders))
        frag.write_text(expand(fragment, shaders))
        subprocess.run(['glslangValidator', '-l', *defines, vert, frag], check=True)
        print(f'PASS {fragment.stem}', flush=True)

out = root / 'timer-source'
out.mkdir(exist_ok=True)
macros = ''.join(f'#define MC_RENDER_STAGE_{name} {stages[name]}\n'
                 for name in ('SKY', 'SUNSET', 'STARS'))
for name, snapshot in [('control', 'control-shaders'), ('weather', 'shaders')]:
    source = root / snapshot
    for stage, ext in [('vsh', 'vert'), ('fsh', 'frag')]:
        version, body = expand(source / f'gbuffers_skybasic.{stage}', source).split('\n', 1)
        (out / f'{name}.{ext}').write_text(version + '\n' + macros + body)
PY
