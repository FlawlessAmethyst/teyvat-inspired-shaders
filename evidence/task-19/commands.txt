from pathlib import Path
from subprocess import check_output
import sys

root = Path(sys.argv[1])

def crop(path, region):
    return check_output(['magick', str(path), '-crop', region, '+repage',
                         '-alpha', 'off', '-depth', '8', 'RGB:-'])

shore = root / 'water-01/shore'
for region in ('640x80+960+30', '640x80+960+1340'):
    a, b = [crop(shore / f'wave-{frame}.png', region) for frame in 'ab']
    mae = sum(abs(x - y) for x, y in zip(a, b, strict=True)) / len(a)
    print(f'{region} mean absolute RGB code difference {mae:.6f}')
    assert (mae > 0) == region.endswith('+30')

for label, region in (('dry deck', '640x80+960+1340'),
                      ('glass over sky', '100x25+1850+475')):
    a, b = [crop(root / f'{mode}-01/basin/native.png', region)
            for mode in ('control', 'water')]
    assert a == b, f'{label} differs across packs'
    print(f'Static {label} RGB matches across both packs')
PY

# Summarize two independent sessions per mode and scene in milliseconds.
python3 - "$root/runs-reviewed.csv" <<'PY'
import csv
from statistics import mean
import sys

with open(sys.argv[1], newline='') as file:
    rows = list(csv.DictReader(file))
for scene in ('basin', 'basin-rain'):
    groups = {}
    for mode in ('sodium', 'control', 'water'):
        group = [row for row in rows if row['scene'] == scene and row['batch'].startswith(mode)]
        assert len(group) == 2
        loops = [float(row['mean_client_loop_ms']) for row in group]
        p95 = [float(row['p95_client_loop_ms']) for row in group]
        groups[mode] = loops
        print(scene, mode, 'mean', round(mean(loops), 6), 'runs', loops, 'p95', p95)
    deltas = [a - b for a, b in zip(groups['water'], groups['control'], strict=True)]
    print(scene, 'water minus TASK-18', round(mean(deltas), 6), 'runs', deltas)
PY
