# Run from the repository root with the dedicated GenshinShaderDev instance stopped.
# The session.sh calls require scoped desktop execution per
# docs/minecraft-environment.md. Preserve the original TASK-10 through TASK-14 artifacts.
set -euo pipefail
root=${1:-artifacts/task-14-repeat}
test ! -e "$root"
mc_dir=$(python3 -c 'import sys; sys.path.insert(0, "tools"); import minecraft; print(minecraft.GAME)')
scene_pack="$mc_dir/saves/TASK-03 Visual Scenes/datapacks/task03"
diff -rq scripts/scenes/data "$scene_pack/data"
cmp scripts/scenes/pack.mcmeta "$scene_pack/pack.mcmeta"

mkdir -p "$root/variants/"{ambient,sky}

sed "s|shaderpack/shaders/|$root/variants/ambient/shaders/|" evidence/task-13/pack.sha256 | sha256sum -c -
sed "s|shaderpack/shaders/|$root/variants/sky/shaders/|" evidence/task-14/pack.sha256 | sha256sum -c -
python3 - "$root" <<'PY'
from pathlib import Path
from tempfile import TemporaryDirectory
import subprocess
import sys
sys.path.insert(0, 'evidence/task-12')
from check_glsl import expand

source = Path(sys.argv[1]) / 'variants/sky/shaders'
with TemporaryDirectory(prefix='task-14-glsl-') as temp:
    for fragment in sorted(source.glob('*.fsh')):
        vertex = fragment.with_suffix('.vsh')
        vert = Path(temp) / (vertex.stem + '.vert')
        frag = Path(temp) / (fragment.stem + '.frag')
        vert.write_text(expand(vertex, source))
        frag.write_text(expand(fragment, source))
        subprocess.run(['glslangValidator', '-l', '-DMC_RENDER_STAGE_SKY=1',
                        '-DMC_RENDER_STAGE_SUNSET=2', vert, frag], check=True)
PY
python3 evidence/task-14/replay.py prepare "$root"

# its batch under another name, then repeat that session. For a full restore,
