# Expand Iris includes before standalone GLSL compilation and linking.
python3 - "$root" <<'PY'
from pathlib import Path
import re
import subprocess
import sys
from tempfile import TemporaryDirectory

def expand(path, shaders):
    return re.sub(r'^#include "/([^"]+)"$', lambda match: expand(shaders / match[1], shaders),
                  path.read_text(), flags=re.MULTILINE)

with TemporaryDirectory(prefix="task-11-glsl-") as temp:
    shipped = Path(sys.argv[1]) / "variants/graded/shaders"
    diagnostic = Path(sys.argv[1]) / "variants/diagnostic/shaders"
    fragments = sorted(shipped.glob("*.fsh")) + [diagnostic / "final.fsh"]
    for fragment in fragments:
        vertex = fragment.with_suffix(".vsh")
        vert = Path(temp) / (vertex.stem + ".vert")
        frag = Path(temp) / (fragment.stem + ".frag")
        vert.write_text(expand(vertex, vertex.parent))
        frag.write_text(expand(fragment, fragment.parent))
        subprocess.run(["glslangValidator", "-l", vert, frag], check=True)
        print(f"PASS {fragment.parent}/{fragment.stem}")
PY

# Record output endpoint counts on the F2 images, in R/G/B order.
python3 - "$root" <<'PY'
from pathlib import Path
import subprocess
import sys
import numpy as np

root = Path(sys.argv[1])
for scene in ('daylight', 'forest', 'sunset', 'night', 'cave'):
    for batch in ('neutral-01', 'graded-01'):
        image = root / batch / scene / 'native.png'
        raw = subprocess.check_output(['magick', image, '-alpha', 'off', '-depth', '8', 'rgb:-'])
        rgb = np.frombuffer(raw, dtype=np.uint8).reshape(-1, 3)
        print(scene, batch, 'zero', np.sum(rgb == 0, axis=0).tolist(),
              '255', np.sum(rgb == 255, axis=0).tolist())
PY
