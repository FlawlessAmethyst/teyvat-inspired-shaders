# Run from the repository root. Desktop operations require approved scoped
# execution as documented in docs/minecraft-environment.md. Run sessions serially.
# Use a fresh artifact root for a replay.
root=artifacts/task-17
mc_dir=$(python3 -c 'import sys; sys.path.insert(0,"tools"); import minecraft; print(minecraft.GAME)')
python3 tools/minecraft.py stop --timeout 60
sha256sum --quiet -c evidence/task-16/pack.sha256
sed 's|shaderpack/shaders/|artifacts/task-16/variants/shadows/shaders/|' \
    evidence/task-16/pack.sha256 | sha256sum --quiet -c -
find "$mc_dir/saves/TASK-03 Visual Scenes" -type f -print0 | sort -z | \
    xargs -0 sha256sum > /tmp/task17-source-world.sha256
diff -rq scripts/scenes/data "$mc_dir/saves/TASK-03 Visual Scenes/datapacks/task03/data"
python3 evidence/task-17/replay.py prepare "$root"
cp /tmp/task17-source-world.sha256 "$root/source-world.sha256"
diff -rq "$root/shaders" artifacts/task-16/variants/shadows/shaders

bash evidence/task-17/session.sh "$root" sodium-01
# Inspect the first world-list row and verify it reads TASK-17 Prototype Scenes.
magick "$root/sodium-01/world-list.png" -crop 800x250+880+100 +repage /tmp/task17-world-row.png
cp "$root/sodium-01/world-list.png" "$root/world-list-reference.png"
bash evidence/task-17/session.sh "$root" sodium-01 open
bash evidence/task-17/session.sh "$root" prototype-01
bash evidence/task-17/session.sh "$root" prototype-02
bash evidence/task-17/session.sh "$root" sodium-02

# After the fourth save and exit:
python3 evidence/task-17/replay.py restore "$root"
sha256sum --quiet -c "$root/source-world.sha256"
find "$mc_dir/saves/TASK-03 Visual Scenes" -type f -print0 | sort -z | \
    xargs -0 sha256sum | cmp "$root/source-world.sha256" -
for path in options.txt config/{iris.properties,sodium-options.json,sodium-extra-options.json,dynamic_fps.json}; do
    cmp "$mc_dir/$path" "$root/before/${path##*/}"
done
test ! -e "$mc_dir/saves/TASK-17 Prototype Scenes"
test ! -e "$mc_dir/shaderpacks/TASK-17-prototype"
sha256sum --quiet -c evidence/task-16/pack.sha256
diff -rq shaderpack/shaders "$root/shaders"

python3 evidence/task-17/index_runs.py "$root" > "$root/runs.csv"
python3 evidence/task-17/index_runs.py "$root" > "$root/reindexed.csv"
cmp "$root/runs.csv" "$root/reindexed.csv"
# The checkpoint's reviewed index is evidence/task-17/runs.csv.
bash evidence/task-17/comparisons.sh "$root"
python3 -m unittest discover -s scripts
python3 -m py_compile evidence/task-17/{replay,index_runs}.py
bash -n evidence/task-17/session.sh
bash -n evidence/task-17/comparisons.sh
python3 scripts/verify_references.py
flatpak info org.prismlauncher.PrismLauncher | rg 'Version:|Branch:|ID:'
