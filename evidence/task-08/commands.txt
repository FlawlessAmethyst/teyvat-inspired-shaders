#!/usr/bin/env bash
# Run from repository root; documentation validation, no Minecraft invocation.
set -euo pipefail
python3 - <<'PY'
from pathlib import Path
import re
from urllib.parse import unquote
paths = [Path(p) for p in ('docs/architecture/shaderpack_design.md',
    'docs/STATUS.md', 'docs/DECISIONS.md', 'docs/TEST_LOG.md')]
links = 0
for path in paths:
    text = path.read_text()
    assert not any(line.rstrip() != line for line in text.splitlines()), path
    for target in re.findall(r'\]\(([^)]+)\)', text):
        if '://' in target or target.startswith('#'):
            continue
        target = unquote(target.split('#')[0])
        assert (path.parent / target).exists(), (path, target)
        links += 1
s = paths[0].read_text()
for target in range(1, 8):
    assert f'| S{target} |' in s
for item in ('RGBA16F', 'depthtex2', 'separateAo=false', 'PARTIAL',
             'TASK-08-SPIKE', 'not run', 'TASK-09', 'TASK-16', 'TASK-19'):
    assert item in s, item
assert not list(Path('shaderpack').rglob('*.vsh'))
assert not list(Path('shaderpack').rglob('*.fsh'))
assert not list(Path('shaderpack').rglob('*.glsl'))
assert 'Current completed task: **TASK-08 — DONE**' in paths[1].read_text()
print(f'PASS: {len(paths)} documents, {links} local links, whitespace, S1–S7 coverage, gate/status markers; no GLSL introduced.')
PY
python3 scripts/verify_references.py
