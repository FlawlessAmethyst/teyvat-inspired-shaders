import csv
from statistics import mean
import sys

with open(sys.argv[1], newline='') as source:
    rows = list(csv.DictReader(source))
for scene in ('night', 'cave'):
    measurements = {}
    for mode in ('sodium', 'control', 'lighting'):
        runs = [row for row in rows if row['scene'] == scene and row['batch'].startswith(mode)]
        assert len(runs) == 2
        values = [float(row['mean_client_loop_ms']) for row in runs]
        p95 = [float(row['p95_client_loop_ms']) for row in runs]
        measurements[mode] = values
        print(f'{scene} {mode}: mean {mean(values):.6f} ms, run means {values}, p95 {p95}')
    control, candidate = (measurements[mode] for mode in ('control', 'lighting'))
    delta = [b - a for a, b in zip(control, candidate, strict=True)]
    print(f'{scene} TASK-22 minus TASK-21: {mean(delta):+.6f} ms, run deltas {delta}')
PY
