# Link all pairs and expand the exact retained sky source for the GPU harness.
python3 - "$root" <<'PY'
from pathlib import Path
from tempfile import TemporaryDirectory
import subprocess
import sys
sys.path.insert(0, 'evidence/task-12')
from check_glsl import expand

root = Path(sys.argv[1])
stages = {'SKY': 1, 'SUNSET': 2, 'STARS': 6, 'HAND_SOLID': 16, 'CLOUDS': 20,
          'RAIN_SNOW': 21, 'HAND_TRANSLUCENT': 23}
defines = [f'-DMC_RENDER_STAGE_{name}={value}' for name, value in stages.items()]
source = root / 'shaders'
with TemporaryDirectory(prefix='task-18-glsl-') as temp:
    for fragment in sorted(source.glob('*.fsh')):
        vertex = fragment.with_suffix('.vsh')
        vert = Path(temp) / (vertex.stem + '.vert')
        frag = Path(temp) / (fragment.stem + '.frag')
        vert.write_text(expand(vertex, source))
        frag.write_text(expand(fragment, source))
        subprocess.run(['glslangValidator', '-l', *defines, vert, frag], check=True)

out = root / 'timer-source'
out.mkdir(exist_ok=True)
defines = ''.join(f'#define MC_RENDER_STAGE_{name} {stages[name]}\n'
                  for name in ('SKY', 'SUNSET', 'STARS'))
for name, snapshot in [('control', 'control-shaders'), ('clouds', 'shaders'), ('trial', 'shaders-trial')]:
    source = root / snapshot
    if name == 'trial' and not source.is_dir():
        continue
    for stage, ext in [('vsh', 'vert'), ('fsh', 'frag')]:
        version, body = expand(source / f'gbuffers_skybasic.{stage}', source).split('\n', 1)
        (out / f'{name}.{ext}').write_text(version + '\n' + defines + body)
PY

cc -std=c11 -O2 -Wall -Wextra -Werror evidence/task-18/sky_timer.c -o "$root/sky_timer" -lEGL -lGL -lm
