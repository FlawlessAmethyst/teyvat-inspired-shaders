# Run from the repository root with GenshinShaderDev stopped.
# Desktop calls use the scoped execution described in docs/minecraft-environment.md.
set -euo pipefail
root=${1:-artifacts/task-16-repeat}
test ! -e "$root"
mc_dir=$(python3 -c 'import sys; sys.path.insert(0,"tools"); import minecraft; print(minecraft.GAME)')
source_world="$mc_dir/saves/TASK-03 Visual Scenes"
abs_root=$(realpath -m "$root")

mkdir -p "$root/variants/shadows"
sed 's|shaderpack/shaders/|artifacts/task-15/variants/fog/shaders/|' \
    evidence/task-15/pack.sha256 | sha256sum -c - > "$root/control-check.txt"
sha256sum -c evidence/task-16/pack.sha256
cp -a artifacts/task-15/variants/fog "$root/variants/"
cp -a shaderpack/shaders "$root/variants/shadows/"
(cd "$source_world" && find . -type f ! -name session.lock -print0 | sort -z | xargs -0 sha256sum) > "$root/source-world.sha256"
python3 evidence/task-16/replay.py prepare "$root"
python3 evidence/task-16/replay.py stage "$root"

# The world-list reference must show TASK-16 Shadow Scenes in the first row.
# On a fresh machine run the first session without the following copy. Inspect
# its world-list.png, copy it to world-list-reference.png if the first row
# matches, then stop the client and archive the menu-only batch before retry.
cp artifacts/task-16/world-list-reference.png "$root/world-list-reference.png"
for batch in sodium-01 fog-01 shadows-01 shadows-02 fog-02 sodium-02 shadows-motion; do
    bash evidence/task-16/session.sh "$root" "$batch"
done

# On interruption, stop the client gracefully before retrying or restoring.
# Preserve any partial batch under another name. Skip the fixture-build trigger
# only if the log already confirms TASK16_READY_FIXTURE in this world copy.
for mode in fog shadows; do
    diff -rq "$root/variants/$mode/shaders" "$mc_dir/shaderpacks/TASK-16-$mode/shaders"
done
python3 evidence/task-16/replay.py restore "$root"
(cd "$source_world" && sha256sum -c "$abs_root/source-world.sha256")
test ! -e "$mc_dir/shaderpacks/TASK-16-fog"
test ! -e "$mc_dir/shaderpacks/TASK-16-shadows"
test ! -e "$mc_dir/saves/TASK-16 Shadow Scenes"

python3 - <<'PY'
from pathlib import Path
from tempfile import TemporaryDirectory
import subprocess
import sys
sys.path.insert(0, 'evidence/task-12')
from check_glsl import expand

source = Path('shaderpack/shaders')
defines = ('-DMC_RENDER_STAGE_SKY=1', '-DMC_RENDER_STAGE_SUNSET=2',
           '-DMC_RENDER_STAGE_HAND_SOLID=16', '-DMC_RENDER_STAGE_CLOUDS=20',
           '-DMC_RENDER_STAGE_RAIN_SNOW=21', '-DMC_RENDER_STAGE_HAND_TRANSLUCENT=23')
with TemporaryDirectory(prefix='task-16-glsl-') as temp:
    for fragment in sorted(source.glob('*.fsh')):
        vertex = fragment.with_suffix('.vsh')
        vert = Path(temp) / (vertex.stem + '.vert')
        frag = Path(temp) / (fragment.stem + '.frag')
        vert.write_text(expand(vertex, source))
        frag.write_text(expand(fragment, source))
        subprocess.run(['glslangValidator', '-l', *defines, vert, frag], check=True)
        print(f'PASS {vertex.stem}')
PY

for batch in sodium-01 fog-01 shadows-01 shadows-02 fog-02 sodium-02; do
    latest=("$root/$batch"/logs-*/latest.log)
    debug=("$root/$batch"/logs-*/debug.log)
    test "${#latest[@]}" -eq 1
    test "$(rg -c 'Stopped tick profiling' "${latest[0]}")" -eq 2
    rg -qF 'All dimensions are saved' "${latest[0]}"
    if [[ "$batch" != sodium-* ]]; then
        rg -qF "Using shaderpack: TASK-16-${batch%%-*}" "${latest[0]}"
        rg -qF 'Creating pipeline for dimension minecraft:overworld' "${latest[0]}"
        rg -q 'Render target settings for colortex0: .*requestedFormat=RGBA16F, clear=true' "${debug[0]}"
    fi
    if rg -ni 'ShaderCompileException|ProgramLinkException|(failed|could not) to (compile|link) shader|failed to create .*pipeline|falling back to vanilla shading' "${latest[0]}"; then
        exit 1
    fi
    for scene in forest sunset; do
        recording="$root/$batch/runs/$scene/recording"
        crop=100x66+1230+1140
        if [[ "$scene" == sunset ]]; then crop=100x60+1100+894; fi
        cmp <(magick "$recording/before.png" -crop "$crop" +repage -alpha off -depth 8 rgb:-) \
            <(magick "$recording/after.png" -crop "$crop" +repage -alpha off -depth 8 rgb:-)
    done
done
