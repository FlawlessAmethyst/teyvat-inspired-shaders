import csv
from statistics import mean
import sys

with open(sys.argv[1], newline='') as file:
    rows = list(csv.DictReader(file))
for scene in ('daylight', 'dense'):
    means = {}
    for mode in ('sodium', 'control', 'vegetation'):
        group = [row for row in rows if row['scene'] == scene and row['batch'].startswith(mode)]
        assert len(group) == 2
        loops = [float(row['mean_client_loop_ms']) for row in group]
        p95 = [float(row['p95_client_loop_ms']) for row in group]
        means[mode] = loops
        print(f'{scene} {mode} mean={mean(loops):.6f} ms runs={loops} p95={p95}')
    deltas = [a - b for a, b in zip(means['vegetation'], means['control'], strict=True)]
    print(f'{scene} vegetation minus TASK-19: {mean(deltas):+.6f} ms '
          f'({mean(deltas) / mean(means["control"]):+.2%}), runs={deltas}')
PY

# Compare native RGB in selected static patches. Animated sky is outside these regions.
python3 - "$root" <<'PY'
from pathlib import Path
from subprocess import check_output
import sys

root = Path(sys.argv[1])

def crop(path, region):
    return check_output(['magick', str(path), '-crop', region, '+repage',
                         '-alpha', 'off', '-depth', '8', 'RGB:-'])

for mode in ('vegetation', 'control', 'sodium'):
    for scene, region in (('daylight', '110x52+2190+485'), ('dense', '105x100+1280+1120')):
        patches = [
            crop(root / f'{mode}-{session}' / 'runs' / scene / 'recording' / f'{frame}.png', region)
            for session in ('01', '02') for frame in ('before', 'after')
        ]
        assert all(patch == patches[0] for patch in patches[1:]), (mode, scene, 'static camera patch changed')
        print(f'{mode}/{scene}: static camera patch matches across both profile sessions')
PY

import csv
from pathlib import Path
from statistics import mean
from subprocess import check_output
import sys

root = Path(sys.argv[1])
patches = (
    ('daylight', 'grass', '450x140+1020+1220', False),
    ('daylight', 'stone brick wall', '110x52+2190+485', True),
    ('forest', 'oak crown', '100x90+1470+330', False),
    ('forest', 'oak trunk', '28x66+1490+610', True),
    ('dense', 'plains oak leaves', '105x100+355+233', False),
    ('dense', 'jungle oak leaves', '105x100+2155+233', False),
    ('dense', 'lit stone path', '105x100+1280+1120', True),
    ('dense', 'shadowed stone path', '23x50+1275+839', True),
)
writer = csv.writer(sys.stdout, lineterminator='\n')
writer.writerow(('scene', 'patch', 'region', 'byte_identical',
                 'control_r', 'control_g', 'control_b',
                 'vegetation_r', 'vegetation_g', 'vegetation_b'))
for scene, label, region, identical in patches:
    control, vegetation = [
        check_output(['magick', str(root / f'{mode}-01' / scene / 'native.png'),
                      '-crop', region, '+repage', '-alpha', 'off', '-depth', '8', 'RGB:-'])
        for mode in ('control', 'vegetation')
    ]
    assert (control == vegetation) == identical, (scene, label)
    values = [f'{mean(data[channel::3]):.2f}' for data in (control, vegetation)
              for channel in range(3)]
    writer.writerow((scene, label, region, identical, *values))
PY

# Check leaf/sky coverage on two lower crown edges against the cloud-free horizon.
python3 - "$root" <<'PY'
from pathlib import Path
from subprocess import check_output
import sys

root = Path(sys.argv[1])
for region in ('230x150+25+450', '180x140+2290+415'):
    masks = []
    for mode in ('control', 'vegetation'):
        image = root / f'{mode}-01/dense/native.png'
        rgb = check_output(['magick', str(image), '-crop', region, '+repage',
                            '-alpha', 'off', '-depth', '8', 'RGB:-'])
        masks.append(bytes(blue >= 150 for blue in rgb[2::3]))
    assert masks[0] == masks[1] and 0 < sum(masks[0]) < len(masks[0]), region
    print(f'{region}: identical leaf/sky coverage across {len(masks[0])} pixels')
PY
