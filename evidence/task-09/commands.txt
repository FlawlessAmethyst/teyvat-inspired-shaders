# Standalone compatibility GLSL syntax/link check with Iris includes expanded.
python3 - <<'PY'
from pathlib import Path
import re
import subprocess
from tempfile import TemporaryDirectory

shaders = Path("shaderpack/shaders")
def expand(path):
    return re.sub(r'^#include "/([^"]+)"$', lambda m: expand(shaders / m[1]),
                  path.read_text(), flags=re.MULTILINE)

with TemporaryDirectory(prefix="task-09-glsl-") as temp:
    for fragment in sorted(shaders.glob("*.fsh")):
        vertex = fragment.with_suffix(".vsh")
        vertex_output = Path(temp) / (vertex.stem + ".vert")
        fragment_output = Path(temp) / (fragment.stem + ".frag")
        vertex_output.write_text(expand(vertex))
        fragment_output.write_text(expand(fragment))
        subprocess.run(["glslangValidator", "-l", vertex_output, fragment_output],
                       check=True, capture_output=True, text=True)
        print(f"PASS glslangValidator -l {vertex.name} {fragment.name}")
PY
