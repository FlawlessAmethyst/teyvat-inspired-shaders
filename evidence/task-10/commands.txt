# Run from the repository root. Desktop commands need the scoped execution
# boundary in docs/minecraft-environment.md. Run one desktop command at a time.
# The original run used artifacts/task-10. Use a fresh root for a replay.
artifact_root="artifacts/task-10-repeat"
neutral_shaders="artifacts/task-11/variants/neutral/shaders"
python3 tools/minecraft.py window
sed "s|shaderpack/shaders/|$neutral_shaders/|" evidence/task-10/pack.sha256 |
  sha256sum -c - &&
  python3 evidence/task-10/configure.py --artifacts-dir "$artifact_root" \
    --shaders-dir "$neutral_shaders" prepare

mc_dir=$(python3 -c 'import sys; sys.path.insert(0,"tools"); import minecraft; print(minecraft.GAME)')
diff -r "$neutral_shaders" "$mc_dir/shaderpacks/TASK-10-neutral/shaders"
diff -rq scripts/scenes/data "$mc_dir/saves/TASK-03 Visual Scenes/datapacks/task03/data"

# Repeat this block sequentially, with the following mode, batch values.
# sodium sodium-01
# neutral neutral-01
# neutral neutral-02
# sodium sodium-02

# Inspect before and after image grids in mode, repeat, scene order.
for frame in before after; do
  images=()
  for batch in sodium-01 neutral-01 neutral-02 sodium-02; do
    for scene in daylight forest cave; do
      images+=("$artifact_root/$batch/runs/$scene/recording/$frame.png")
    done

  magick montage "${images[@]}" -thumbnail 480x270 -tile 3x -geometry +2+2 \
    "$artifact_root/$frame-grid.png"
