# Run from the repository root with GenshinShaderDev stopped.
# session.sh and the interactive menu calls need scoped desktop execution
# as described in docs/minecraft-environment.md.
set -euo pipefail
root=${1:-artifacts/task-15-repeat}
test ! -e "$root"
mc_dir=$(python3 -c 'import sys; sys.path.insert(0,"tools"); import minecraft; print(minecraft.GAME)')
source_world="$mc_dir/saves/TASK-03 Visual Scenes"
abs_root=$(realpath -m "$root")
mkdir -p "$root/variants/sky"
cp -a artifacts/task-14/variants/sky/shaders "$root/variants/sky/"
sed "s|shaderpack/shaders/|$root/variants/sky/shaders/|" evidence/task-14/pack.sha256 | sha256sum -c -
(cd "$source_world" && find . -type f ! -name session.lock -print0 | sort -z | xargs -0 sha256sum) > "$root/source-world.sha256"
python3 evidence/task-15/replay.py prepare "$root"

# Reuse the inspected TASK-15 first-row title mask. On a fresh machine,
# invoke the first session without this copy. Inspect its world-list.png,
# copy it to world-list-reference.png if TASK-15 Fog Scenes is the first row,
# then stop the game, archive the menu-only batch and retry sodium-01.
cp artifacts/task-15/world-list-reference.png "$root/world-list-reference.png"
for batch in sodium-01 sky-01; do
    bash evidence/task-15/session.sh "$root" "$batch"
done

sha256sum -c evidence/task-15/pack.sha256
python3 - <<'PY'
from pathlib import Path
from tempfile import TemporaryDirectory
import subprocess
import sys
sys.path.insert(0, 'evidence/task-12')
from check_glsl import expand

source = Path('shaderpack/shaders')
defines = ('-DMC_RENDER_STAGE_SKY=1', '-DMC_RENDER_STAGE_SUNSET=2',
           '-DMC_RENDER_STAGE_HAND_SOLID=16', '-DMC_RENDER_STAGE_CLOUDS=20',
           '-DMC_RENDER_STAGE_RAIN_SNOW=21', '-DMC_RENDER_STAGE_HAND_TRANSLUCENT=23')
with TemporaryDirectory(prefix='task-15-glsl-') as temp:
    for fragment in sorted(source.glob('*.fsh')):
        vertex = fragment.with_suffix('.vsh')
        vert = Path(temp) / (vertex.stem + '.vert')
        frag = Path(temp) / (fragment.stem + '.frag')
        vert.write_text(expand(vertex, source))
        frag.write_text(expand(fragment, source))
        subprocess.run(['glslangValidator', '-l', *defines, vert, frag], check=True)
        print(f'PASS {vertex.stem}')
PY
mkdir "$root/variants/fog"

python3 evidence/task-15/replay.py stage-fog "$root"

# If interrupted, stop the scoped client gracefully, preserve the incomplete
# batch under a different name and repeat it. To restore immediately, stop
# the client and run replay.py restore with the original artifact root.
for variant in sky fog; do
    diff -rq "$root/variants/$variant/shaders" "$mc_dir/shaderpacks/TASK-15-$variant/shaders"
done
python3 evidence/task-15/replay.py restore "$root"
(cd "$source_world" && sha256sum -c "$abs_root/source-world.sha256")
test ! -e "$mc_dir/shaderpacks/TASK-15-sky"
test ! -e "$mc_dir/shaderpacks/TASK-15-fog"
test ! -e "$mc_dir/saves/TASK-15 Fog Scenes"

    if rg -ni 'TASK03_READY_build|ShaderCompileException|ProgramLinkException|(failed|could not) to (compile|link) shader|failed to create .*pipeline|falling back to vanilla shading' "${latest[0]}"; then

    done
done
