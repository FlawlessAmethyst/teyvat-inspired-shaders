(
    python3 evidence/task-28/replay.py prepare "$root"
    finish() {
        result=$?
        python3 tools/minecraft.py --artifacts-dir "$root" stop --timeout 60 || return
        python3 evidence/task-28/replay.py restore "$root" || return
        exit "$result"
    }
    trap finish EXIT
    cc -std=c11 -O2 -Wall -Wextra -Werror evidence/task-28/motion_input.c -o "$root/motion-input"

    test "$(rg -c '^PASS ' "$root/static-link.txt")" = 13
    bash evidence/task-28/session.sh "$root"
